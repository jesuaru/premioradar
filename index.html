<script>

const KEY = 'premioRadarNotasAude';

function loadNotes() {

  try { return JSON.parse(localStorage.getItem(KEY) || '[]'); }

  catch { return []; }

}

function saveNotes(notes) {

  localStorage.setItem(KEY, JSON.stringify(notes));

}

function esc(s) {

  return String(s || '').replace(/[&<>"']/g, c => ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c]));

}

function render() {

  const list = document.getElementById('notesList');

  const notes = loadNotes();

  if (!notes.length) {

    list.innerHTML = '<p class="empty">Todavía no hay notas guardadas.</p>';

    return;

  }

  list.innerHTML = notes.map((n, i) => `

    <div class="note-item">

      <b>${esc(n.date)}</b>

      ${n.url ? `<a href="${esc(n.url)}" target="_blank">${esc(n.url)}</a>` : ''}

      ${n.text ? `<p>${esc(n.text)}</p>` : ''}

      <button class="del" data-del="${i}">Eliminar</button>

    </div>

  `).join('');

}

document.getElementById('addNote').addEventListener('click', () => {

  const url = document.getElementById('noteUrl').value.trim();

  const text = document.getElementById('noteText').value.trim();

  if (!url && !text) return;

  const notes = loadNotes();

  notes.unshift({

    url,

    text,

    date: new Date().toLocaleString('es', { dateStyle: 'short', timeStyle: 'short' })

  });

  saveNotes(notes);

  document.getElementById('noteUrl').value = '';

  document.getElementById('noteText').value = '';

  render();

});

document.getElementById('notesList').addEventListener('click', (e) => {

  const btn = e.target.closest('[data-del]');

  if (!btn) return;

  if (!confirm('¿Eliminar esta nota?')) return;

  const notes = loadNotes();

  notes.splice(Number(btn.dataset.del), 1);

  saveNotes(notes);

  render();

});

render();

</script>
